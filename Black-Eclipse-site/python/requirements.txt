# Browser Automation Runtime — structured browser control behind Eclipse's own
# provider/runtime abstraction (the model never sees a raw Playwright object).
# OPTIONAL in effect: a missing Playwright makes the browser toolset report an
# honest BROWSER_UNAVAILABLE (it is never registered), exactly like the

# BROWSER_UNINSTALLED until it is present.
# Pinned to the 1.62 series so it shares the SAME chromium build the app's Node
# Playwright already uses (they cache under one ms-playwright directory).
playwright>=1.65.2,<1.66

# Computer Use Engine (native desktop) — OPTIONAL in effect. Every backend here
# is honest: a missing one degrades to UNAVAILABLE/reason, never a fake desktop.
# * Pillow: image capture (GDI fallback encoding) + vision frame encoding.
# * uiautomation (Windows): the rich accessibility tree (UIA). Without it the
#   driver falls back to the Win32 child-window baseline.
# * winsdk (Windows): the OS-native OCR engine (Windows.Media.Ocr) — no external
#   binary, no download, no cloud.
# Cross-platform Pillow is installed by default; the Windows-only backends carry
# markers so they are not attempted on macOS/Linux.
Pillow>=13.0.0
uiautomation>=2.0.29; sys_platform == "win32"
winsdk>=1.0.0b10; sys_platform == "win32"

# Excel / PowerPoint creation works out of the box.

# * openpyxl     — SpreadsheetML
# * python-pptx  — PresentationML (pulls lxml/Pillow/XlsxWriter, all present)

# PyAV 15+ removed the `metadata_errors` keyword that faster-whisper passes to
# av.open(), which broke local transcription with "open() got an unexpected
# keyword argument". Pin the range that works.
#
# Upper bound is 14, not 15: PyAV 14.4.0 ships an sdist ONLY (no wheels on any
# platform), so `pip install -r requirements.lock` tries to compile it and dies
# twice over - Linux without FFmpeg headers ("pkg-config could not find
# libraries [avformat, ...]"), and with them on a setuptools manifest error.
# 13.x publishes manylinux/macos/windows wheels for cp312, so the stack installs
# from a wheel. Raise this only if 14.x starts shipping wheels again.
av>=19.0.2,<20
